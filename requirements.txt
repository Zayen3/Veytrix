from metadata import extract_metadata
from analyzer import analyze_privacy, print_privacy_report
from sanitizer import (
    sanitize_image,
    verify_sanitization,
    remove_gps_data,
    remove_device_data,
    remove_timestamp_data,
    remove_software_data,
)
from gps import check_gps_data

from gui import launch_app


if __name__ == "__main__":
    launch_app()


def main():
    """Run the Veytrix metadata privacy tool."""

    file_path = input("Enter the path of an image: ").strip()

    try:
        # Extract metadata
        metadata = extract_metadata(file_path)

        # Analyze privacy risks
        analysis = analyze_privacy(metadata)

        # Display privacy report
        print_privacy_report(metadata, analysis)

        # Display sanitization options
        print("\nSANITIZATION OPTIONS")
        print("1. Remove GPS location data only")
        print("2. Remove device information only")
        print("3. Remove timestamp information only")
        print("4. Remove software information only")
        print("5. Remove all EXIF metadata")
        print("6. Cancel")

        choice = input("\nChoose an option (1/2/3/4/5/6): ").strip()

        if choice == "1":
            cleaned_path = remove_gps_data(file_path)

            print("\n" + "=" * 55)
            print("VEYTRIX - GPS SANITIZATION")
            print("=" * 55)

            print(f"\nOriginal file: {metadata['file_name']}")
            print(f"Cleaned file: {cleaned_path.name}")
            print("Status: GPS location data removed.")

            # Verify GPS removal
            cleaned_metadata = extract_metadata(cleaned_path)

            print("\n--- SANITIZATION VERIFICATION ---")

            if check_gps_data(cleaned_metadata["exif_data"]):
                print(
                    "⚠ Verification warning: "
                    "GPS metadata may still remain."
                )
            else:
                print(
                    "✓ Verification successful: "
                    "GPS metadata was removed."
                )

        elif choice == "2":
            cleaned_path = remove_device_data(file_path)

            print("\n" + "=" * 55)
            print("VEYTRIX - DEVICE SANITIZATION")
            print("=" * 55)

            print(f"\nOriginal file: {metadata['file_name']}")
            print(f"Cleaned file: {cleaned_path.name}")
            print("Status: Device information removed.")

            # Verify device information removal
            cleaned_metadata = extract_metadata(cleaned_path)

            print("\n--- SANITIZATION VERIFICATION ---")

            if (
                cleaned_metadata["make"] is None
                and cleaned_metadata["model"] is None
            ):
                print(
                    "✓ Verification successful: "
                    "Device information was removed."
                )
            else:
                print(
                    "⚠ Verification warning: "
                    "Some device information may still remain."
                )

        elif choice == "3":
            cleaned_path = remove_timestamp_data(file_path)

            print("\n" + "=" * 55)
            print("VEYTRIX - TIMESTAMP SANITIZATION")
            print("=" * 55)

            print(f"\nOriginal file: {metadata['file_name']}")
            print(f"Cleaned file: {cleaned_path.name}")
            print("Status: Timestamp information removed.")

            # Verify all common timestamp fields were removed
            cleaned_metadata = extract_metadata(cleaned_path)

            print("\n--- SANITIZATION VERIFICATION ---")

            if (
                cleaned_metadata["capture_date"] is None
                and cleaned_metadata["original_date"] is None
                and cleaned_metadata["digitized_date"] is None
            ):
                print(
                    "✓ Verification successful: "
                    "Timestamp information was removed."
                )
            else:
                print(
                    "⚠ Verification warning: "
                    "Some timestamp information may still remain."
                )

        elif choice == "4":
            cleaned_path = remove_software_data(file_path)

            print("\n" + "=" * 55)
            print("VEYTRIX - SOFTWARE SANITIZATION")
            print("=" * 55)

            print(f"\nOriginal file: {metadata['file_name']}")
            print(f"Cleaned file: {cleaned_path.name}")
            print("Status: Software information removed.")

            # Verify software information removal
            cleaned_metadata = extract_metadata(cleaned_path)

            print("\n--- SANITIZATION VERIFICATION ---")

            if cleaned_metadata["software"] is None:
                print(
                    "✓ Verification successful: "
                    "Software information was removed."
                )
            else:
                print(
                    "⚠ Verification warning: "
                    "Software information may still remain."
                )

        elif choice == "5":
            cleaned_path = sanitize_image(file_path)

            print("\n" + "=" * 55)
            print("VEYTRIX - METADATA SANITIZATION")
            print("=" * 55)

            print(f"\nOriginal file: {metadata['file_name']}")
            print(f"Cleaned file: {cleaned_path.name}")
            print("Status: All EXIF metadata removed.")

            # Verify full sanitization
            verified = verify_sanitization(cleaned_path)

            print("\n--- SANITIZATION VERIFICATION ---")

            if verified:
                print(
                    "✓ Verification successful: "
                    "No EXIF metadata found."
                )
            else:
                print(
                    "⚠ Verification warning: "
                    "EXIF metadata may still remain."
                )

        elif choice == "6":
            print("\nNo changes were made to the original image.")

        else:
            print("\nInvalid choice. No changes were made.")

    except FileNotFoundError as error:
        print(f"\nError: {error}")

    except RuntimeError as error:
        print(f"\nError: {error}")


if __name__ == "__main__":
    main()
